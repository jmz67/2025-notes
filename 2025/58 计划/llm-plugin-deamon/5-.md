
# 5.2 构建 Go Web 服务器
---


```go
package main
// 引入需要使用的包
import (
    "fmt"    // 用于格式化输入输出
    "log"    // 用于记录日志
    "net/http" // 用于构建HTTP相关功能
    "strings" // 用于字符串操作
)

// sayHello函数处理HTTP请求，w是用于向客户端写入响应的对象，r是包含请求信息的对象
func sayHello(w http.ResponseWriter, r *http.Request) {
    // 解析请求中的参数，默认情况下参数不会自动解析
    _ = r.ParseForm() 
    // 将解析后的参数信息输出到服务器端的打印信息中
    fmt.Println(r.Form) 
    // 输出请求的URL路径
    fmt.Println("Path: ", r.URL.Path) 
    // 输出请求的主机名
    fmt.Println("Host: ", r.Host) 
    // 遍历请求参数
    for k, v := range r.Form {
        // 输出参数的键
        fmt.Println("key:", k) 
        // 将参数的值切片连接成字符串并输出
        fmt.Println("val:", strings.Join(v, "")) 
    }
    // 将问候信息写入到响应对象w，其中包含从请求参数中获取的"name"值
    _, _ = fmt.Fprintf(w, "Hello Web, %s!", r.Form.Get("name")) 
}

func main() {
    // 设置根路径"/"的请求处理函数为 sayHello
    http.HandleFunc("/", sayHello) 
    // 设置监听的端口为8080，nil 表示使用默认的 ServeMux
    err := http.ListenAndServe(":8080", nil) 
    if err != nil {
        // 如果监听启动失败，记录严重错误日志并终止程序
        log.Fatal("ListenAndServe: ", err) 
    }
}
```
