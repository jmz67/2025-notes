```
plugin_attr:
  # prometheus:
  #   export_addr:
  #     ip: "0.0.0.0"
  #     port: 9091
  prometheus:
    export_addr:
      ip: "0.0.0.0"
      port: 9091
    metrics:
      http_status:
        extra_labels:
          - upstream_addr: $upstream_addr
          - route_name: $route_name
      http_latency:
        extra_labels:
          - upstream_addr: $upstream_addr
          - route_name: $route_name
```
